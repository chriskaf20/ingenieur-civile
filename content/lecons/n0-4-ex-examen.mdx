import { composantes, convertirAngle, cosD, ecrituresPente, EXEMPLE_FERME, ferme, penteDepuis, rectDeuxCotes, toiture, triangleACA, triangleCAC } from '@/lib/calc/trigo'
import { SEUIL_TEXTE } from '@/lib/progression/types'
import { grandeur, nombre } from '@/lib/format'

export const gon = convertirAngle(1, 'gon', 'deg')
export const rad2 = convertirAngle(2, 'rad', 'deg')
export const h125 = rectDeuxCotes(12, 5)
export const p25 = ecrituresPente(0.25)
export const t21 = ecrituresPente(penteDepuis(2, 'n'))
export const c135 = cosD(135)
export const f20 = composantes(20, 30)
export const kashi = triangleCAC(6, 10, 40)
export const sinus = triangleACA(50, 60, 12)
export const toit = toiture({ L: 10, longueur: 1, pente: 25, debord: 0 })
export const fer = ferme({ ...EXEMPLE_FERME, L: 9 })
