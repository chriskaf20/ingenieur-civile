import { bogue, concentrationDepH, diagnosticCarbonatation, dureeCarbonatation, LIMITES_GACHAGE, massesReaction, masseMolaire, reaction, vitessePenetration } from '@/lib/calc/chimie'
import { SEUIL_TEXTE } from '@/lib/progression/types'

export const q2 = masseMolaire('CaCO3')

export const q7 = bogue({ CaO: 65, SiO2: 21, Al2O3: 5, Fe2O3: 3, SO3: 1 })

export const q10b = dureeCarbonatation(30, q10.K)
