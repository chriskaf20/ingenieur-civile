import { E_S } from '@/lib/calc/ba/materiaux'
import { analyser, ecrireDim, formule, DIM } from '@/lib/calc/dimensions'
import { MATERIAUX, poidsPrismeMateriau } from '@/lib/calc/materiauxCourants'
import { poidsDalle } from '@/lib/calc/ordres'
import { changerPrefixe, convertir, poidsDeMasse, pressionMoyenne } from '@/lib/calc/unites'
import { SEUIL_TEXTE } from '@/lib/progression/types'
import { grandeur, nombre, nombreCourt } from '@/lib/format'

export const es = changerPrefixe(E_S, 'M', 'G').valeur
export const cm2 = convertir(1.5, 'm²', 'cm²').valeur
export const litres = convertir(2.4, 'm³', 'L').valeur
export const hect = convertir(3500, 'm²', 'ha').valeur
export const sac = poidsDeMasse(25)
export const sem = pressionMoyenne(450, 1.5, 1.5)
export const pou = poidsPrismeMateriau('ba', 6, 0.25, 0.5)
export const dal = poidsDalle(1, 1, 0.18).charge
export const faux = analyser(formule('faux-poids'))
export const bar = convertir(1.8, 'bar', 'kPa').valeur
export const tf = convertir(20, 'tf', 'kN').valeur
export const pouce = convertir(1.5, 'in', 'mm').valeur
