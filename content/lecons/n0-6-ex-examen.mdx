import { depuisPolaire, elingues, fois, forceCable, matHaubane, moment2D, norme, scalaire, sommeForces, travail } from '@/lib/calc/vecteurs'
import { decomposerSurVersant } from '@/lib/calc/trigo'
import { SEUIL_TEXTE } from '@/lib/progression/types'
import { grandeur, nombre } from '@/lib/format'

export const vec = (v, d = 2) => `(${nombre(v[0], d)} ; ${nombre(v[1], d)})`
export const c1 = forceCable({ xA: 0, yA: 4, xB: 3, yB: 0, T: 15, echelle: 1 })
export const n2 = norme([-5, 12])
export const s3 = sommeForces([{ F: 8, theta: 0 }, { F: 6, theta: 90 }])
export const s4 = sommeForces([{ F: 20, theta: 0 }, { F: 20, theta: 120 }])
export const p5 = fois(-3, [2, 5])
export const f6 = depuisPolaire(12, 150)
export const t7 = decomposerSurVersant(20, 25)
export const e8 = elingues({ P: 30, alpha1: 45, alpha2: 45 })
export const ps9 = scalaire([4, 1], [-1, 4])
export const w10 = travail(depuisPolaire(3, 60), [10, 0])
export const m11 = moment2D([0, 0], [2, 5], [4, -3])
export const h12 = matHaubane({ h: 9, r: 12, H: 3, psi: 270, phi0: 90, marge: 0 })
export const t12 = Math.max(...h12.haubans.map((h) => h.T))
